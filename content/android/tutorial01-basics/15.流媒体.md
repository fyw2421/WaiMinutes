# 一、SoundPool 播放音频（游戏，段时间）

1. 创建 SoundPool 对象
2. 使用 SoundPool 对象加载音乐数据
3. 播放加载好的音乐

```
public class MainActivity extends Activity {

    private SoundPool soundPool;

    private int[] id = new int[5];

    private int i = 0;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        // 1. 创建SoundPool对象
        this.soundPool = new SoundPool(5, AudioManager.STREAM_MUSIC, 0);

        // 2.加载音乐数据
        int id0 = this.soundPool.load(this, R.raw.angry, 1);
        //异步加载
        int id1 = this.soundPool.load(this, R.raw.fall, 1);
        int id2 = this.soundPool.load(this, R.raw.p_belly1, 1);
        int id3 = this.soundPool.load(this, R.raw.cymbal, 1);
        int id4 = this.soundPool.load(this, R.raw.p_eat, 1);
        // this.soundPool.load("/mnt/sdcard/vvv/ss.mp3", 1);
        id[0] = id0;
        id[1] = id1;
        id[2] = id2;
        id[3] = id3;
        id[4] = id4;
    }

    //(int soundID, float leftVolume, float rightVolume, int priority, int loop, float rate)
    public void run(View v) {
        // 3. 播放加载好的音乐
        this.soundPool.play(this.id[i], 0.5f, 1.0f, 0, 0, 1.0f);
        i++;
        if(i==5) {
            i=0;
        }
    }
}
```

# 二、MediaPlayer 播放音频

1. 创建 MediaPlayer，封装音乐文件来源
2. 准备播放音乐（停止、预备）
3. 播放音乐
4. 停止，释放，制空

```
public class MainActivity extends Activity {

    private MediaPlayer mediaPlayer=null;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
    }

    public void run(View v) {
        //通过XML文件设置事件监听 android:onClick="run".....
        Button btn = (Button) this.findViewById(R.id.run);
        btn.setClickable(false);
        //设置按钮不可点击
        btn.setText("正在播放中......");
        if(this.mediaPlayer != null) {
            return;
        }
        //.创建MediaPlayer，封装音乐文件来源
        this.mediaPlayer = MediaPlayer.create(this, R.raw.rain);
        this.mediaPlayer.stop();
        //停止音乐
        try {
            this.mediaPlayer.prepare();
            //预备
        } catch (IOException e) {
            e.printStackTrace();
        }
        this.mediaPlayer.start();
        /*放音乐*/
    }

    public void stop(View v) {
        this.mediaPlayer.stop();
        /*停止音乐*/
        this.mediaPlayer.release();
        this.mediaPlayer = null;
        Button btn = (Button) this.findViewById(R.id.run);
        btn.setClickable(true);
        //设置按钮可点击
        btn.setText("播放音乐");
    }

    public void test() {
        this.mediaPlayer.pause();
        int time = this.mediaPlayer.getDuration();
        //得到音频播放的总时间(ms)
        int time1 = this.mediaPlayer.getCurrentPosition();
        //播放到了多小毫秒
        this.mediaPlayer.seekTo(20*1000);
        //进程指定在20秒钟处
        this.mediaPlayer.setLooping(false);
        boolean b = this.mediaPlayer.isLooping();
        //循环播放
        boolean b2 = this.mediaPlayer.isPlaying();
        //表示音乐是否在播放中
    }
}

# 三、SeekBar 进度条

```
public class MainActivity extends Activity {

    private SeekBar seekBar;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        this.setContentView(R.layout.activity_main);
        this.seekBar = (SeekBar) this.findViewById(R.id.seekBar1);
        this.seekBar.setMax(1000);
        // 设置进度条最大长度
        this.setListener();
    }

    private void setListener() {
        this.seekBar.setOnSeekBarChangeListener(
                new OnSeekBarChangeListener() {
                    @Override
                    public void onStopTrackingTouch(SeekBar seekBar) {
                        System.out.println("onStopTrackingTouch");
                    }

                    @Override
                    public void onStartTrackingTouch(SeekBar seekBar) {
                        System.out.println("onStartTrackingTouch");
                    }

                    @Override
                    public void onProgressChanged(SeekBar seekBar, int progress, boolean fromUser) {
                        System.out.println("" + progress + ":" + fromUser);
                    }
                });
    }

    public void start(View v) {
        //通过XML文件设置事件监听 android:onClick="start".....
        new Thread() {
            @Override
            public void run() {
                for (int i = 0; i <= 1000; i++) {
                    seekBar.setProgress(i);
                    // 获取进度条当前进度长度
                    try {
                        Thread.sleep(10);
                    } catch (InterruptedException e) {
                        e.printStackTrace();
                    }
                }
            }
        }.start();
    }
}

# 五、RecordSound 音频录制

```
public class RecordSound extends Activity implements OnClickListener {

    // 定义界面上的两个按钮
    ImageButton record, stop;

    // 系统的音频文件
    File soundFile;

    MediaRecorder mRecorder;

    @Override
    public void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.main);

        // 获取程序界面中的两个按钮
        record = (ImageButton) findViewById(R.id.record);
        stop = (ImageButton) findViewById(R.id.stop);

        // 为两个按钮的单击事件绑定监听器
        record.setOnClickListener(this);
        stop.setOnClickListener(this);
    }

    @Override
    public void onDestroy() {
        if (soundFile != null && soundFile.exists()) {
            // 停止录音
            mRecorder.stop();
            // 释放资源
            mRecorder.release();
            mRecorder = null;
        }
        super.onDestroy();
    }

    @Override
    public void onClick(View v) {
        switch (v.getId()) {
            // 单击录音按钮
            case R.id.record:
                if(!new File("/mnt/sdcard").exists()){
                    //如果  /mnt/sdcard  不存在
                    Toast.makeText(RecordSound.this, "SD卡不存在，请插入SD卡！",Toast.LENGTH_SHORT).show();
                    return;
                }
                try {
                    // 创建保存录音的音频文件
                    soundFile = new File("/mnt/sdcard/mm"+System.currentTimeMillis()+".amr");
                    mRecorder = new MediaRecorder();
                    // 设置录音的声音来源
                    mRecorder.setAudioSource(MediaRecorder.AudioSource.MIC);
                    // 设置声音编码的格式
                    mRecorder.setAudioEncoder(MediaRecorder.AudioEncoder.AMR_NB);
                    mRecorder.setOutputFile(soundFile.getAbsolutePath());//  "/mnt/sdcard/mm.amr"
                    // 设置录制的声音的输出格式（必须在设置声音编码格式之前设置）
                    mRecorder.setOutputFormat(MediaRecorder.OutputFormat.THREE_GPP);
                    mRecorder.prepare();
                    //准备工作
                    // 开始录音
                    mRecorder.start(); // ①
                } catch (Exception e) {
                    e.printStackTrace();
                }
                break;

            // 单击停止按钮
            case R.id.stop:
                if (soundFile != null && soundFile.exists()) {
                    // 停止录音
                    mRecorder.stop(); // ②
                    // 释放资源
                    mRecorder.release(); // ③
                    mRecorder = null;
                }
                break;
        }
    }
}

# 六、视频录制

```
public class RecordVideo extends Activity implements OnClickListener {

    // 程序中的两个按钮
    ImageButton record, stop;

    // 系统的视频文件
    File videoFile;

    MediaRecorder mRecorder;

    // 显示视频预览的SurfaceView
    SurfaceView sView;

    // 记录是否正在进行录制
    private boolean isRecording = false;

    @Override
    public void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.main);

        // 获取程序界面中的两个按钮
        record = (ImageButton) findViewById(R.id.record);
        stop = (ImageButton) findViewById(R.id.stop);
        // 让stop按钮不可用。
        stop.setEnabled(false);

        // 为两个按钮的单击事件绑定监听器
        record.setOnClickListener(this);
        stop.setOnClickListener(this);

        // 获取程序界面中的SurfaceView
        sView = (SurfaceView) this.findViewById(R.id.sView);

        // 设置Surface不需要自己维护缓冲区
        SurfaceHolder holder = sView.getHolder();
        holder.setType(SurfaceHolder.SURFACE_TYPE_PUSH_BUFFERS);

        // 设置分辨率
        holder.setFixedSize(320, 280);

        // 设置该组件让屏幕不会自动关闭
        holder.setKeepScreenOn(true);
    }

    @Override
    public void onClick(View source) {
        switch (source.getId()) {
            // 单击录制按钮
            case R.id.record:
                //判断是否存在"/mnt/sdcard"卡
                //  boolean b = !Environment.getExternalStorageState().equals(android.os.Environment.MEDIA_MOUNTED);
                boolean b = new File("/mnt/sdcard").exists();
                if (!b){
                    Toast.makeText(RecordVideo.this, "SD卡不存在，请插入SD卡！",Toast.LENGTH_SHORT).show();
                    return;
                }
                try {
                    // 创建保存录制视频的视频文件
                    videoFile = new File("/mnt/sdcard"+"/fs"+System.currentTimeMillis()+".mp4");
                    // 创建MediaPlayer对象
                    mRecorder = new MediaRecorder();
                    mRecorder.reset();
                    //重置
                    // 设置从麦克风采集声音
                    mRecorder.setAudioSource(MediaRecorder.AudioSource.MIC);
                    // 设置从摄像头采集图像
                    mRecorder.setVideoSource(MediaRecorder.VideoSource.CAMERA);
                    // 设置视频文件的输出格式
                    // 必须在设置声音编码格式、图像编码格式之前设置
                    mRecorder.setOutputFormat(MediaRecorder.OutputFormat.MPEG_4);
                    // 设置声音编码的格式
                    mRecorder.setAudioEncoder(MediaRecorder.AudioEncoder.DEFAULT);
                    // 设置图像编码的格式
                    mRecorder.setVideoEncoder(MediaRecorder.VideoEncoder.MPEG_4_SP);
                    mRecorder.setVideoSize(320, 280);
                    // 每秒 4帧
                    mRecorder.setVideoFrameRate(4);
                    //设置视频保存的文件路径
                    mRecorder.setOutputFile(videoFile.getAbsolutePath());
                    // 指定使用SurfaceView来预览视频
                    mRecorder.setPreviewDisplay(sView.getHolder().getSurface()); // ①
                    mRecorder.prepare();
                    // 开始录制
                    mRecorder.start();
                    System.out.println("---recording---");
                    // 让record按钮不可用。
                    record.setEnabled(false);
                    // 让stop按钮可用。
                    stop.setEnabled(true);
                    isRecording = true;
                } catch (Exception e) {
                    e.printStackTrace();
                }
                break;

            // 单击停止按钮
            case R.id.stop:
                // 如果正在进行录制
                if (isRecording) {
                    // 停止录制
                    mRecorder.stop();
                    // 释放资源
                    mRecorder.release();
                    mRecorder = null;
                    // 让record按钮可用。
                    record.setEnabled(true);
                    // 让stop按钮不可用。
                    stop.setEnabled(false);
                }
                break;
        }
    }
}
