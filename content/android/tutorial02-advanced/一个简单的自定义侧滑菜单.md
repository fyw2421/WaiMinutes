写出这个控件不难，但里面有几个方法和概念很重要，很多地方都会用到，怕自己也会忘，注释的很清楚。

```
public class QQcelamenu extends HorizontalScrollView {

    private LinearLayout mWrapper;//contain menu and content View
    private ViewGroup mMenuLayout;
    private ViewGroup mContentLayout;
    private int Swidth;
    private int menuRightMargin=50;
    private boolean once=false,PositionChanged=false;

    public QQcelamenu(Context context) {
        this(context,null);
    }

    public QQcelamenu(Context context, AttributeSet attrs) {
        this(context, attrs,0);
    }

    public QQcelamenu(Context context, AttributeSet attrs, int defStyle) {
        super(context, attrs, defStyle);
        Swidth=context.getResources().getDisplayMetrics().widthPixels;
        menuRightMargin=(int) TypedValue.applyDimension(TypedValue.COMPLEX_UNIT_DIP, 50, context.getResources().getDisplayMetrics());
    }

    @Override
    protected void onMeasure(int widthMeasureSpec, int heightMeasureSpec) {
        if (!once) {
            mWrapper=(LinearLayout) getChildAt(0);
            mMenuLayout=(ViewGroup) mWrapper.getChildAt(0);
            mContentLayout=(ViewGroup) mWrapper.getChildAt(1);
            mMenuLayout.getLayoutParams().width=Swidth-menuRightMargin;
            mContentLayout.getLayoutParams().width=Swidth;
            once=true;
        }
        super.onMeasure(widthMeasureSpec, heightMeasureSpec);
    }

    @Override
    protected void onLayout(boolean changed, int l, int t, int r, int b) {
        // TODO Auto-generated method stub
        super.onLayout(changed, l, t, r, b);
        if (!PositionChanged) {
            //滚动条要移动的方向，正值时，滚动条右移，视图左移;  负值，滚动条左移，视图右移
            this.scrollTo(mMenuLayout.getMeasuredWidth(), 0);//把菜单隐藏
            PositionChanged=true;
        }
    }

    @Override
    public boolean onTouchEvent(MotionEvent ev) {
        // TODO Auto-generated method stub
        switch (ev.getAction()) {
            case MotionEvent.ACTION_UP:
                //返回的是控件的left值，left，top，right，bottom这几个值都是相对于其父容器的值
                int scrollx=getScrollX();
                if (scrollx>mMenuLayout.getMeasuredWidth()/2) {
                    this.smoothScrollTo(mMenuLayout.getMeasuredWidth(), 0);
                } else {
                    this.smoothScrollTo(0, 0);
                }
                return true;
            default:
                break;
        }
        return super.onTouchEvent(ev);
    }

}
```
